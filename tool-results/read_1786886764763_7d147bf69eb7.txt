    61→interface Employee {

    69→}
    70→
    71→interface PurchaseOrder {
    72→  id: string; orderNumber: string; supplierName: string; date: string; expectedDelivery: string;
    73→  total: number; status: 'draft' | 'sent' | 'approved' | 'received' | 'cancelled' | 'converted'
    74→}
    75→
    76→interface JournalEntry {

    78→  totalDebit: number; totalCredit: number; status: 'draft' | 'posted' | 'reversed'
    79→}
    80→

    86→interface Check {
